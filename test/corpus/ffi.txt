ext configlib {
  go: "github.com/example/configlib"
  ts: "@example/configlib"

  struct go_config { Host: string, DevHost: string, Env: string }
  struct ts_config { host: string, token: string }

  extern load(service: string, region: string): app_config {
    go {
      call: "Load"(service, region)
      yields: (cfg: go_config)
      returns: app_config {
        endpoint: match .cfg.Env { "prod" => .cfg.Host, _ => .cfg.DevHost }
        token: .cfg.Token
      }

    ts {
      call: "load"(region, service)
      yields: (cfg: ts_config)
      returns: app_config { endpoint: .cfg.host, token: .cfg.token }

  go: "github.com/example/store"
  ts: "@example/store"

  struct go_body { ID: string, Text: string }
  struct ts_opts { id: string, retries: i32 }

  extern fetch(id: string): fetched {
    go {
      call: "Fetch"(id)
      yields: (err: error, body: go_body, ok: bool)
      returns: fetched { id: .body.ID, text: .body.Text, verified: .ok }
      errors: { "ErrBusy" => overloaded, "ErrGone" => not_found }
    }
    ts {
      call: "fetch"(ts_opts { id: id, retries: 3 })
      errors: { "BUSY" => overloaded }
    }
  }
}
